------------------------------------------------------------
---------- МОДУЛЬНАЯ ФАЙЛОВАЯ СТРУКТУРА -----------

|__ .env            # Реальные пароли базы данных и Redis.
|__ .env.example    # Шаблон переменных окружения без паролей.
|
|__ config/            # Скрытая директория конфигурации.
|   |__ redis.php      # Redis (читает данные из .env).
|   |__ db.php         # БД (читает данные из .env).
|   |__ chat.php       # Настройки админки (не в БД).
|   |__ topics.php     # Темы с переводами для (l.php / list.php).
|
|__ functions/   # Скрытая директория функций.
|   |__ lang.php                # функция t() + определение языка.
|   |__ function-question.php   # Функции чата вопросов.
|   |__ function-list.php       # Функции списка вопросов.
|   |__ function-answer.php     # Функции чата ответов.
|   |__ function-settings.php   # Функции аккаунта с настройками.
|   |__ function-enter.php      # Функции авторизации/регистрации.
|   |__ function-chat.php       # Функции общего чата.
|
|__ lang/                  # Скрытая директория переводов.
|   |__ _common/      # Общие строки: меню, кнопки, темы, ошибки.
|   |   |__ ru.php
|   |   |__ en.php
|   |   |__ es.php
|   |   |__ ...
|   |
|   |__ q/            # Строки главной (Спросить).
|   |   |__ ru.php
|   |   |__ en.php
|   |
|   |__ l/            # Строки списка (Ответить).
|   |   |__ ru.php
|   |   |__ en.php
|   |
|   |__ a/            # Строки детали Ответов.
|   |   |__ ru.php
|   |   |__ en.php
|   |
|   |__ s/            # Строки аккаунта/настроек.
|   |   |__ ru.php
|   |   |__ en.php
|   |
|   |__ e/            # Строки входа/регистрации.
|       |__ ru.php
|       |__ en.php
|
|__ modules/           # Скрытая директория шаблонов модулей.
|   |__ header.php     # Шаблон шапки.
|   |__ panel.php      # Шаблон боковой панели.
|   |__ question.php   # Шаблон модуля чата для отправки вопроса.
|   |__ list.php       # Шаблон модуля списка вопросов.
|   |__ answer.php     # Шаблон модуля с выбранным вопросом.
|   |__ settings.php   # Шаблон модуля аккаунта с настройками.
|   |__ enter.php      # Шаблон модуля авторизации/регистрации.
|   |__ chat.php       # Шаблон модуля общего чата.
|
|__ public_html/  # Публичная директория. 
    |__ q.php     # Главная страница с чатом для отправки вопроса.
    |__ l.php     # Страница со списком вопросов.
    |__ a.php     # Страница с выбранным вопросом.
    |__ s.php     # Страница аккаунта с настройками.
    |__ e.php     # Страница авторизации/регистрации.
    |__ c.php     # Страница общего чата.
    |__ out.php   # Выйти.
    |__ .htaccess # Конфигурации htaccess.
